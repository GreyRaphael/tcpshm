function(tcpshm_add_test target src)
  add_executable(${target} ${src})
  target_link_libraries(${target} PRIVATE tcpshm::tcpshm)
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W3)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endfunction()

tcpshm_add_test(unit_spsc_test                 unit_spsc_test.cpp)
tcpshm_add_test(unit_ptcp_test                 unit_ptcp_test.cpp)
tcpshm_add_test(integration_handshake_test     integration_handshake_test.cpp)
tcpshm_add_test(integration_server_client_test integration_server_client_test.cpp)
